104 function eof(s::LibuvStream)
105 bytesavailable(s) > 0 && return false
106 wait_readnb(s, 1)
107 # This function is race-y if used from multiple threads, but we guarantee
108 # it to never return true until the stream is definitively exhausted
109 # and that we won't return true if there's a readerror pending (it'll instead get thrown).
110 # This requires some careful ordering here (TODO: atomic loads)
111 bytesavailable(s) > 0 && return false
112 open = isreadable(s) # must precede readerror check
113 s.readerror === nothing || throw(s.readerror)
114 return !open
115 end

392 function wait_readnb(x::LibuvStream, nb::Int)
393 # fast path before iolock acquire
394 bytesavailable(x.buffer) >= nb && return
395 open = isopen(x) && x.status != StatusEOF # must precede readerror check
396 x.readerror === nothing || throw(x.readerror)
397 open || return
398 iolock_begin()
399 # repeat fast path after iolock acquire, before other expensive work
400 bytesavailable(x.buffer) >= nb && (iolock_end(); return)
401 open = isopen(x) && x.status != StatusEOF
402 x.readerror === nothing || throw(x.readerror)
403 open || (iolock_end(); return)
404 # now do the "real" work
405 oldthrottle = x.throttle
406 preserve_handle(x)
407 lock(x.cond)
408 try
409 while bytesavailable(x.buffer) < nb
410 x.readerror === nothing || throw(x.readerror)
411 isopen(x) || break
412 x.status == StatusEOF && break
413 x.throttle = max(nb, x.throttle)
414 start_reading(x) # ensure we are reading
415 iolock_end()
416 wait(x.cond)
417 unlock(x.cond)
418 iolock_begin()
419 lock(x.cond)
420 end
421 finally
422 if isempty(x.cond)
423 stop_reading(x) # stop reading iff there are currently no other read clients of the stream
424 end
425 if oldthrottle <= x.throttle <= nb
426 # if we're interleaving readers, we might not get back to the "original" throttle
427 # but we consider that an acceptable "risk", since we can't be quite sure what the intended value is now
428 x.throttle = oldthrottle
429 end
430 unpreserve_handle(x)
431 unlock(x.cond)
432 end
433 iolock_end()
434 nothing
435 end

437 function closewrite(s::LibuvStream)
438 iolock_begin()
439 check_open(s)
440 req = Libc.malloc(_sizeof_uv_shutdown)
441 uv_req_set_data(req, C_NULL) # in case we get interrupted before arriving at the wait call
442 err = ccall(:uv_shutdown, Int32, (Ptr{Cvoid}, Ptr{Cvoid}, Ptr{Cvoid}),
443 req, s, @cfunction(uv_shutdowncb_task, Cvoid, (Ptr{Cvoid}, Cint)))
444 if err < 0
445 Libc.free(req)
446 uv_error("shutdown", err)
447 end
448 ct = current_task()
449 preserve_handle(ct)
450 sigatomic_begin()
451 uv_req_set_data(req, ct)
452 iolock_end()
453 status = try
454 sigatomic_end()
455 wait()::Cint
456 finally
457 # try-finally unwinds the sigatomic level, so need to repeat sigatomic_end
458 sigatomic_end()
459 iolock_begin()
460 ct.queue === nothing || list_deletefirst!(ct.queue::IntrusiveLinkedList{Task}, ct)
461 if uv_req_data(req) != C_NULL
462 # req is still alive,
463 # so make sure we won't get spurious notifications later
464 uv_req_set_data(req, C_NULL)
465 else
466 # done with req
467 Libc.free(req)
468 end
469 iolock_end()
470 unpreserve_handle(ct)
471 end
472 if isopen(s)
473 if status < 0 || ccall(:uv_is_readable, Cint, (Ptr{Cvoid},), s.handle) == 0
474 close(s)
475 end
476 end
477 if status < 0
478 throw(_UVError("shutdown", status))
479 end
480 nothing
481 end

656 function uv_readcb(handle::Ptr{Cvoid}, nread::Cssize_t, buf::Ptr{Cvoid})
657 stream_unknown_type = @handle_as handle LibuvStream
658 nrequested = ccall(:jl_uv_buf_len, Csize_t, (Ptr{Cvoid},), buf)
659 function readcb_specialized(stream::LibuvStream, nread::Int, nrequested::UInt)
660 lock(stream.cond)
661 if nread < 0
662 if nread == UV_ENOBUFS && nrequested == 0
663 # remind the client that stream.buffer is full
664 notify(stream.cond)
665 elseif nread == UV_EOF # libuv called uv_stop_reading already
666 if stream.status != StatusClosing
667 stream.status = StatusEOF
668 notify(stream.cond)
669 if stream isa TTY
670 # stream can still be used by reseteof (or possibly write)
671 elseif !(stream isa PipeEndpoint) && ccall(:uv_is_writable, Cint, (Ptr{Cvoid},), stream.handle) != 0
672 # stream can still be used by write
673 else
674 # underlying stream is no longer useful: begin finalization
675 ccall(:jl_close_uv, Cvoid, (Ptr{Cvoid},), stream.handle)
676 stream.status = StatusClosing
677 end
678 end
679 else
680 stream.readerror = _UVError("read", nread)
681 notify(stream.cond)
682 # This is a fatal connection error
683 ccall(:jl_close_uv, Cvoid, (Ptr{Cvoid},), stream.handle)
684 stream.status = StatusClosing
685 end
686 else
687 notify_filled(stream.buffer, nread)
688 notify(stream.cond)
689 end
690 unlock(stream.cond)
691
692 # Stop background reading when
693 # 1) there's nobody paying attention to the data we are reading
694 # 2) we have accumulated a lot of unread data OR
695 # 3) we have an alternate buffer that has reached its limit.
696 if stream.status == StatusPaused ||
697 (stream.status == StatusActive &&
698 ((bytesavailable(stream.buffer) >= stream.throttle) ||
699 (bytesavailable(stream.buffer) >= stream.buffer.maxsize)))
700 # save cycles by stopping kernel notifications from arriving
701 ccall(:uv_read_stop, Cint, (Ptr{Cvoid},), stream)
702 stream.status = StatusOpen
703 end

1394 throw(ArgumentError("stdin and stderr cannot be the same path"))
1395 end
1396 if same_path(stdout, stdin)
1397 throw(ArgumentError("stdin and stdout cannot be the same path"))
1398 end
1399
1400 new_in , close_in , old_in = resolve(stdin , Base.stdin , "r")
1401 new_out, close_out, old_out = resolve(stdout, Base.stdout, "w")
1402 if same_path(stderr, stdout)
1403 # make sure that in case stderr = stdout = "same/path"
1404 # only a single io is used instead of opening the same file twice
1405 new_err, close_err, old_err = new_out, false, Base.stderr
1406 else
1407 new_err, close_err, old_err = resolve(stderr, Base.stderr, "w")
1408 end
1409
1410 redirect_stdio(; stderr=new_err, stdin=new_in, stdout=new_out)
1411
1412 try
1413 return f()
1414 finally
1415 redirect_stdio(;stderr=old_err, stdin=old_in, stdout=old_out)
1416 close_err && close(new_err)
1417 close_in && close(new_in )
1418 close_out && close(new_out)
1419 end
1420 end
