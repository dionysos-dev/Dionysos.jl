195 function _rewrite_to_jump_logic(x)
196 if Meta.isexpr(x, :call)
197 op = if x.args[1] == :ifelse
198 return Expr(:call, op_ifelse, x.args[2:end]...)
199 elseif x.args[1] == :<
200 return Expr(:call, op_strictly_less_than, x.args[2:end]...)
201 elseif x.args[1] == :>
202 return Expr(:call, op_strictly_greater_than, x.args[2:end]...)
203 elseif x.args[1] == :<=
204 return Expr(:call, op_less_than_or_equal_to, x.args[2:end]...)
205 elseif x.args[1] == :>=
206 return Expr(:call, op_greater_than_or_equal_to, x.args[2:end]...)
207 elseif x.args[1] == :(==)
208 return Expr(:call, op_equal_to, x.args[2:end]...)
209 end
210 elseif Meta.isexpr(x, :||)
211 # Take special care to ensure short-circuiting behavior of operator is
212 # retained. We don't want to evaluate the second argument if the first
213 # is `true`.
214 @assert length(x.args) == 2
215 return Expr(
216 :if,
217 Expr(:call, ===, x.args[1], true),
218 true,
219 Expr(:call, op_or, x.args[1], x.args[2]),
220 )
221 elseif Meta.isexpr(x, :&&)
222 # Take special care to ensure short-circuiting behavior of operator is
223 # retained. We don't want to evaluate the second argument if the first
224 # is `false`.
225 @assert length(x.args) == 2
226 return Expr(
227 :if,
228 Expr(:call, ===, x.args[1], false),
229 false,
230 Expr(:call, op_and, x.args[1], x.args[2]),
231 )
232 elseif Meta.isexpr(x, :comparison)
233 lhs = Expr(:call, x.args[2], x.args[1], x.args[3])
234 rhs = Expr(:call, x.args[4], x.args[3], x.args[5])
235 return Expr(
236 :call,
237 op_and,
238 _rewrite_to_jump_logic(lhs),
239 _rewrite_to_jump_logic(rhs),
240 )
241 end
242 return x
243 end

547 """
548 print_macro_timing_summary([io::IO = stdout], model::GenericModel)
549
550 Print a summary of the runtime of each macro.
551
552 Before calling this method, you must have enabled the macro timing feature using
553 [`set_macro_timing`](@ref).

555 ## Example
556
557 ```jldoctest; filter=[r"Total time inside macros: .+ seconds", r"├.+", r"└.+"]
558 julia> begin

560 set_macro_timing(model, true)

564

566 Total time inside macros: 5.33690e-02 seconds
567 │
568 ├ 2.96490e-02 s [55.55%]
569 │ ├ REPL[8]:3
570 │ └ `@variable(model, x[1:2])`
571 │
572 └ 2.37200e-02 s [44.45%]
573 ├ REPL[8]:4
574 └ `@objective(model, Min, sum(x))`
575 ```
576 """
577 function print_macro_timing_summary(io::IO, model::GenericModel)
578 total_time = sum(values(model.macro_times))
579 times = sort!(collect(model.macro_times); by = last, rev = true)
580 println(io, "Total time inside macros: ", _format_time(total_time))
581 for i in 1:length(times)
582 (source, expr), time = times[i]
583 percent = round(100 * time / total_time; digits = 2)
584 a, b = ifelse(i < length(times), ('├', '│'), ('└', ' '))
585 println(io, "│")
586 println(io, "$a ", _format(time), " s [$percent%]")
587 println(io, "$b ├ $(source.file):$(source.line)")
588 println(io, "$b └ ", replace(_string_summary(expr), "\n" => ""))
589 end
590 return
591 end
