300 """
301 outer_keys(d::AbstractDoubleDict)
302
303 Return an iterator over the outer keys of the `AbstractDoubleDict` `d`.
304 Each outer key is a `Tuple{Type,Type}` so that a double loop can
305 be easily used:
306 ```julia
307 for (F, S) in DoubleDicts.outer_keys(dict)
308 for (k, v) in dict[F, S]
309 # ...
310 end
311 end
312 ```
313
314 For performance, it is recommended that the inner loop lies in a separate
315 function to guarantee type-stability.
316 Some outer keys `(F, S)` might lead to an empty `dict[F, S]`.
317 If you want only nonempty `dict[F, S]`, use [`nonempty_outer_keys`](@ref).
318 """
319 outer_keys(d::AbstractDoubleDict) = keys(d.dict)
320
321 """
322 nonempty_outer_keys(d::AbstractDoubleDict)
323
324 Return a vector of outer keys of the `AbstractDoubleDict` `d`.
325
326 Only outer keys that have a nonempty set of inner keys will be returned.
327
328 Each outer key is a `Tuple{Type,Type}` so that a double loop can
329 be easily used
330 ```julia
331 for (F, S) in DoubleDicts.nonempty_outer_keys(dict)
332 for (k, v) in dict[F, S]
333 # ...
334 end
335 end
336 For performance, it is recommended that the inner loop lies in a separate
337 function to guarantee type-stability.
338
339 If you want an iterator of all current outer keys, use [`outer_keys`](@ref).
340 ```
341 """
342 function nonempty_outer_keys(d::AbstractDoubleDict)
343 return Base.Iterators.Filter(
344 fs -> length(d[fs[1], fs[2]]) > 0,
345 outer_keys(d),
