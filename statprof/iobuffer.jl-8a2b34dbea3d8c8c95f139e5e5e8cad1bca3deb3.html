36 """
37 IOBuffer([data::AbstractVector{UInt8}]; keywords...) -> IOBuffer
38
39 Create an in-memory I/O stream, which may optionally operate on a pre-existing array.
40
41 It may take optional keyword arguments:
42 - `read`, `write`, `append`: restricts operations to the buffer; see `open` for details.
43 - `truncate`: truncates the buffer size to zero length.
44 - `maxsize`: specifies a size beyond which the buffer may not be grown.
45 - `sizehint`: suggests a capacity of the buffer (`data` must implement `sizehint!(data, size)`).
46
47 When `data` is not given, the buffer will be both readable and writable by default.
48
49 # Examples
50 ```jldoctest
51 julia> io = IOBuffer();
52
53 julia> write(io, "JuliaLang is a GitHub organization.", " It has many members.")
54 56
55
56 julia> String(take!(io))
57 "JuliaLang is a GitHub organization. It has many members."
58
59 julia> io = IOBuffer(b"JuliaLang is a GitHub organization.")
60 IOBuffer(data=UInt8[...], readable=true, writable=false, seekable=true, append=false, size=35, maxsize=Inf, ptr=1, mark=-1)
61
62 julia> read(io, String)
63 "JuliaLang is a GitHub organization."
64
65 julia> write(io, "This isn't writable.")
66 ERROR: ArgumentError: ensureroom failed, IOBuffer is not writeable
67
68 julia> io = IOBuffer(UInt8[], read=true, write=true, maxsize=34)
69 IOBuffer(data=UInt8[...], readable=true, writable=true, seekable=true, append=false, size=0, maxsize=34, ptr=1, mark=-1)
70
71 julia> write(io, "JuliaLang is a GitHub organization.")
72 34
73
74 julia> String(take!(io))
75 "JuliaLang is a GitHub organization"
76
77 julia> length(read(IOBuffer(b"data", read=true, truncate=false)))
78 4
79
80 julia> length(read(IOBuffer(b"data", read=true, truncate=true)))
81 0
82 ```
83 """

85 data::AbstractVector{UInt8};

88 append::Union{Bool,Nothing}=nothing,
89 truncate::Union{Bool,Nothing}=nothing,
90 maxsize::Integer=typemax(Int),
91 sizehint::Union{Integer,Nothing}=nothing)
92 if maxsize < 0
93 throw(ArgumentError("negative maxsize"))
94 end

96 sizehint!(data, sizehint)
97 end

99 buf = GenericIOBuffer(data, flags.read, flags.write, true, flags.append, Int(maxsize))
100 if flags.truncate
101 buf.size = 0
102 end
103 return buf
104 end

458 ensureroom(to, nb)

460 written = Int(min(nb, Int(length(to.data))::Int - ptr + 1))
