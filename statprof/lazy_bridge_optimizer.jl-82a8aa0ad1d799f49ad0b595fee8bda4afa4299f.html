9 """
10 LazyBridgeOptimizer(model::MOI.ModelLike)
11
12 The `LazyBridgeOptimizer` is a bridge optimizer that supports multiple bridges,
13 and only bridges things which are not supported by the internal model.
14
15 Internally, the `LazyBridgeOptimizer` solves a shortest hyper-path problem to
16 determine which bridges to use.
17
18 In general, you should use [`full_bridge_optimizer`](@ref) instead of this
19 constructor because [`full_bridge_optimizer`](@ref) automatically adds a large
20 number of supported bridges.
21
22 See also: [`add_bridge`](@ref), [`remove_bridge`](@ref), [`has_bridge`](@ref)
23 and [`full_bridge_optimizer`](@ref).
24
25 ## Example
26
27 ```jldoctest
28 julia> model = MOI.Bridges.LazyBridgeOptimizer(MOI.Utilities.Model{Float64}());
29
30 julia> MOI.Bridges.add_bridge(model, MOI.Bridges.Variable.FreeBridge{Float64})
31
32 julia> MOI.Bridges.has_bridge(model, MOI.Bridges.Variable.FreeBridge{Float64})
33 true
34 ```
35 """

129 """
130 _reset_bridge_graph(b::LazyBridgeOptimizer)
131
132 Reset the hyper-graph maintained by the [LazyBridgeOptimizer](@ref), but do not
133 remove the bridges that have been added.
134
135 We need to reset the graph whenever we add or delete bridges from `b`, because
136 it might change the optimal hyper-paths. Recomputing from scratch is a lot of
137 work, but it will only happen if we interleave calls to [`add_bridge`](@ref),
138 [`remove_bridge`](@ref), and model builders like [`MOI.add_constraint`](@ref).
139 """
140 function _reset_bridge_graph(b::LazyBridgeOptimizer)
141 empty!(b.variable_node)
142 empty!(b.variable_types)
143 empty!(b.constraint_node)
144 empty!(b.constraint_types)
145 empty!(b.objective_node)
146 empty!(b.objective_types)
147 empty!(b.graph)
148 empty!(b.cached_bridge_type)
149 return
150 end

254 function node(

257 )

259 if (

261 MOI.supports_add_constrained_variable(b.model, S)
262 ) || (
263 S <: MOI.AbstractVectorSet &&
264 MOI.supports_add_constrained_variables(b.model, S)
265 )
266 return VariableNode(0)
267 end
268 # If (S,) is stored in .variable_node, we've already added the node
269 # previously.
270 variable_node = get(b.variable_node, (S,), nothing)
271 if variable_node !== nothing
272 return variable_node
273 end
274 # This is a new (S,). We need to add it to the graph.
275 variable_node = add_node(b.graph, VariableNode)
276 b.variable_node[(S,)] = variable_node
277 push!(b.variable_types, (S,))
278 F = MOI.Utilities.variable_function_type(S)
279 if is_bridged(b, MOI.Reals)
280 # The solver doesn't support adding free variables.
281 FF = _functionized_type(b, F)

284 # node `MOI.Reals` is `1` (that is, it is bridged by
285 # `Variable.FreeBridge`. We also use the functionize bridge which
286 # has cost 1. And we add `+1` as we treat constrained variables as
287 # constraints.

293 )

301 edge = _edge(b, i, Variable.concrete_bridge_type(BT, S))::Edge
