149 function is_diagonal_vectorized_index(index::Base.Integer)
150 # See https://en.wikipedia.org/wiki/Triangular_number#Triangular_roots_and_tests_for_triangular_numbers
151 perfect_square = 1 + 8index
152 return isqrt(perfect_square)^2 == perfect_square
153 end

155 # We have `d*(d+1)/2 = n` so `2n = d^2 + d` hence
156 # `d^2 ≤ 2n < d^2 + 2d + 1 = (d + 1)^2`
157 # this means that `d` is the largest natural number `d`
158 # such that `d^2 ≤ 2n` hence `d = isqrt(2n)`.

167 """
168 trimap(row::Integer, column::Integer)
169
170 Convert between the row and column indices of a matrix, to the linear index of
171 the corresponding element in the triangular representation.
172
173 This is most useful when mapping between `ConeSquare` and `ConeTriangle` sets,
174 for example, as part of an [`MOI.AbstractSymmetricMatrixSetTriangle`](@ref) set.
175
176 !!! note
177 Use [`inverse_trimap`](@ref) for the reverse mapping.
178 """
179 function trimap(row::Integer, column::Integer)
180 if row < column
181 return trimap(column, row)
182 end
183 return div((row - 1) * row, 2) + column
184 end

186 """
187 inverse_trimap(index::Integer)
188
189 Convert between the linear index of a
190 [`MathOptInterface.AbstractSymmetricMatrixSetTriangle`] to the row and column
191 indices of upper triangular part of the corresponding matrix.
192
193 !!! note
194 Use [`trimap`](@ref) for the reverse mapping.
195 """
196 function inverse_trimap(index::Integer)
197 # Because `isqrt` and `div` are monotone functions
198 # because `index` is between `trimap(j - 1, j - 1)` and
199 # `trimap(j, j)`,`
200 # `side_dimension_for_vectorized_dimension(index)` is between `j - 1`
201 # and `j`.
202 j = side_dimension_for_vectorized_dimension(index)
203 n = trimap(j, j)
204 if index <= n
205 j -= 1
206 n = trimap(j, j)
207 end
208 i = index - n
209 return i, j + 1
210 end
