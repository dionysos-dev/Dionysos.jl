192 # Calling `@default_eltye` on `x` isn't sufficient, because the iterator may
193 # skip every element based on the condition. Call it on an identical nested
194 # iterator, but this time without the condition.
195 _default_eltype(x::NestedIterator) = Base.@default_eltype nested(x.iterators...)
196 _default_eltype(x) = Base.@default_eltype x
