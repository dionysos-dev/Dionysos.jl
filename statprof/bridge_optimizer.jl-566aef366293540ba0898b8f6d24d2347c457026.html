83 function is_bridged(
84 b::AbstractBridgeOptimizer,
85 ci::MOI.ConstraintIndex{MOI.VariableIndex,S},
86 ) where {S}
87 # There are a few cases for which we should return `false`:
88 # 1) It was added as variables constrained on creation to `b.model`,
89 # In this case, `is_bridged(b, S)` is `false` and `ci.value >= 0`.
90 # 2) It was added as constraint on a non-bridged variable to `b.model`,
91 # In this case, `is_bridged(b, F, S)` is `false` and `ci.value >= 0`.
92 # and a few cases for which we should return `true`:
93 # 3) It was added with a variable bridge,
94 # In this case, `is_bridged(b, S)` is `true` and `ci.value < 0`.
95 # 4) It was added as constraint on a bridged variable so it was force-bridged,
96 # In this case, `ci.value < 0`.
97 # 5) It was added with a constraint bridge,
98 # In this case, `is_bridged(b, F, S)` is `true` and `ci.value >= 0` (the variable is non-bridged, otherwise, the constraint would have been force-bridged).
99 # So
100 # * if, `ci.value < 0` then it is case 3) or 4) and we return `true`.
101 # * Otherwise,
102 # - if `is_bridged(b, S)` and `is_bridged(b, F, S)` then 1) and 2) are
103 # not possible so we are in case 5) and we return `true`.
104 # - if `!is_bridged(b, F, S)`, then 5) is not possible and we return `false`.
105 # - if `!is_bridged(b, S)` and `is_bridged(b, F, S)`, then it is either case 1)
106 # or 5). They cannot both be the cases as one cannot add two `VariableIndex`
107 # with the same set type on the same variable (this is ensured by
108 # `_check_double_single_variable`). Therefore, we can safely determine
109 # whether it is bridged with `haskey(Constraint.bridges(b), ci)`.
110 return ci.value < 0 || (
111 is_bridged(b, MOI.VariableIndex, S) &&
112 (is_bridged(b, S) || haskey(Constraint.bridges(b), ci))
113 )
114 end

475 function MOI.is_valid(
476 b::AbstractBridgeOptimizer,
477 ci::MOI.ConstraintIndex{F,S},
478 ) where {F,S}
479 if is_bridged(b, ci)
480 if is_variable_bridged(b, ci)
481 v_map = Variable.bridges(b)::Variable.Map
482 return MOI.is_valid(v_map, ci)
483 else
484 # This return value has the potential to be a false positive: it
485 # doesn't discriminate between constraints that the user added, and
486 # constraints that a bridge added that were themselves bridged.
487 #
488 # "Fixing" this particular call has a number of wide reaching
489 # effects because bridges need this to be "true" so that they can
490 # query attributes of the constraint from `b`.
491 #
492 # In most cases a false positive doesn't matter, because we really
493 # do support querying stuff about it. And also the user needs some
494 # way of obtaining the correct index, which they won't have except
495 # by luck/enumeration.
496 #
497 # The main place that this is problematic is when we come to delete
498 # constraints, and in particular VariableIndex constraints, because we
499 # triviallly have their `.value` field from the `.value` of the
500 # VariableIndex.
501 #
502 # Instead of fixing everything though, we implement some extra
503 # checks when deleting, and we leave the false-positive as-is for
504 # now. If you, future reader, hit this comment while debugging, we
505 # might need to revisit this decision.
506 #
507 # x-ref https://github.com/jump-dev/MathOptInterface.jl/issues/2696
508 # x-ref https://github.com/jump-dev/MathOptInterface.jl/issues/2817
509 # x-ref https://github.com/jump-dev/MathOptInterface.jl/pull/2818
510 return haskey(Constraint.bridges(b), ci)
511 end
512 else
513 return MOI.is_valid(b.model, ci)

575 _delete_variables_in_variables_constraints(

577 vis::Vector{MOI.VariableIndex},

580 The point of this function is to delete constraints associated with the
581 variables in `vis`.
582
583 ## Warning
584
585 Because of the false positive potential in
586 `is_valid(::AbstractBridgeOptimizer, MOI.ConstraintIndex)`, we need to ensure

794 function _get_all_including_bridged(
795 b::AbstractBridgeOptimizer,
796 attr::MOI.ListOfVariableIndices,
797 )
798 # `inner_to_outer` is going to map variable indices in `b.model` to their
799 # bridged variable indices. If the bridge adds multiple variables, we need
800 # only to map the first variable, and we can skip the rest. To mark this
801 # distinction, the tail variables are set to `nothing`.
802 map = Variable.bridges(b)
803 inner_to_outer = Dict{MOI.VariableIndex,Union{Nothing,MOI.VariableIndex}}()
804 # These are variables which appear in `b` but do NOT appear in `b.model`.
805 # One reason might be the Zero bridge in which they are replaced by `0.0`.
806 user_only_variables = MOI.VariableIndex[]
807 for (user_variable, bridge) in map
808 variables = MOI.get(bridge, MOI.ListOfVariableIndices())
809 if isempty(variables)
810 # This bridge maps `user_variable` to constants in `b.model`. We
811 # still need to report back `user_variable` to the user. In
812 # addition, it might represent the start of a VectorOfVariables
813 # function, so we may need to report multiple variables.
814 push!(user_only_variables, user_variable)
815 n = Variable.length_of_vector_of_variables(map, user_variable)
816 for i in 1:(n-1)
817 push!(
818 user_only_variables,
819 MOI.VariableIndex(user_variable.value - i),
820 )
821 end
822 else
823 # This bridge maps `user_variable` to a list of `variables`. We need
824 # to swap out only the `first` variable. The others can be flagged
825 # with `nothing`. To simplify the first/tail loop, we set
826 # `first(variables)` twice; first to `nothing` and then to
827 # `user_variable`.
828 for bridged_variable in variables
829 inner_to_outer[bridged_variable] = nothing
830 end
831 inner_to_outer[first(variables)] = user_variable
832 end
833 end
834 # We're about to loop over the variables in `.model`, ordered by when they
835 # were added to the model. We need to return a list of the original
836 # user-space variables in the order that they were added. To do so, we need
837 # to undo any Variable.bridges transformations.
838 ret = MOI.VariableIndex[]
839 for inner_variable in MOI.get(b.model, attr)
840 outer_variable = get(inner_to_outer, inner_variable, missing)
841 # If there is a chain of variable bridges, the `outer_variable` may need
842 # to be mapped.
843 while haskey(inner_to_outer, outer_variable)
844 outer_variable = inner_to_outer[outer_variable]
845 end
846 if ismissing(outer_variable)
847 # inner_variable does not exist in inner_to_outer, which means that
848 # it is not bridged. Pass through unchanged.
849 push!(ret, inner_variable)
850 elseif isnothing(outer_variable)
851 # inner_variable exists in inner_to_outer, but it is set to `nothing`
852 # which means that it is not the first variable in the bridge. Skip
853 # it because it should be hidden from the user.
854 else
855 # inner_variable exists in inner_to_outer. It must be the first
856 # variable in the bridge. Report it back to the user.
857 push!(ret, outer_variable)
858 # `outer_variable` might represent the start of a VectorOfVariables
859 # if multiple user-variables were bridged. Add them all.
860 n = Variable.length_of_vector_of_variables(map, outer_variable)
861 for i in 1:(n-1)
862 push!(ret, MOI.VariableIndex(outer_variable.value - i))
863 end
864 end
865 end
866 # Since these were replaced by constants, we don't actually know when they
867 # were added to the model. Tack them on at the end...
868 #
869 # If you, future reader, find this troublesome, come up with a better
870 # solution.
871 append!(ret, user_only_variables)
872 return ret
873 end

2505 if !Variable.has_bridges(Variable.bridges(b))
2506 return func, set
2507 end

2509 # implement getters for `MOI.ConstraintFunction` and `MOI.ConstraintSet`.
2510 #
2511 # See `unbridged_constraint_function` for more details.
2512 MOI.throw_if_scalar_and_constant_not_zero(func, typeof(set))
2513 f = bridged_function(b, func)::typeof(func)
2514 return MOI.Utilities.normalize_constant(f, set)
2515 end

2521 )

2527
2528 # The purpose of unbridged_constraint_function is to convert the result of
2529 # ConstraintFunction into the un-bridged form. Since variable bridges create
2530 # substitution rules which are scalar functions, for example, `x => y_1 - y_2`, if the
2531 # result is a VariableIndex or an AbstractVectorFunction we can return the
2532 # unbridged function:
2533
2534 function unbridged_constraint_function(
2535 b::AbstractBridgeOptimizer,
2536 func::MOI.VariableIndex,
2537 )
2538 return unbridged_function(b, func)
2539 end
2540
2541 function unbridged_constraint_function(
2542 b::AbstractBridgeOptimizer,
2543 func::MOI.AbstractVectorFunction,
2544 )
2545 return unbridged_function(b, func)
2546 end
2547
2548 # (These are separate methods to avoid ambiguity issues.)
2549
2550 # But if the result is an AbstractScalarFunction, we need to be more careful.
2551
2552 function unbridged_constraint_function(
2553 b::AbstractBridgeOptimizer,
2554 func::MOI.AbstractScalarFunction,
2555 )
2556 # If `b` has no variable bridges, no substitution can occur, so we return
2557 # func.
2558 if !Variable.has_bridges(Variable.bridges(b))
2559 return func
2560 end
2561 # Otherwise, first unbridge the function:
2562 f = unbridged_function(b, func)::typeof(func)
2563 # But now we have to deal with an issue. Something like x in [1, ∞) might
2564 # get bridged into y in R₊, with x => y + 1, so if the original constraint is
2565 # 2x >= 1, the bridged function is 2y >= -1. Unbridging this with y = x - 1
2566 # gives 2x - 2, but we only care about 2x. Where did the -2 come from? It
2567 # was moved into the set. This gets handled separately, so for
2568 # ConstraintFunction it is sufficient to drop any non-zero constant terms.
2569 #
2570 # It's also safe to do this because ScalarFunctionConstantNotZero will be
2571 # thrown if we can't move the constant into the set, and there is a test to
2572 # catch this.
2573 c = MOI.constant(f)
2574 if !iszero(c)
2575 f = MOI.Utilities.operate(-, typeof(c), f, c)
2576 end
2577 return f
2578 end
