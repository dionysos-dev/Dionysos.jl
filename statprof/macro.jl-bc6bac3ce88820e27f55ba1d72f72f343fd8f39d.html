20 parse_macro_arguments(
21 error_fn::Function,
22 args;
23 valid_kwargs::Union{Nothing,Vector{Symbol}} = nothing,
24 num_positional_args::Union{Nothing,Int,UnitRange{Int}} = nothing,
25 )
26
27 Returns a `Tuple{Vector{Any},Dict{Symbol,Any}}` containing the ordered
28 positional arguments and a dictionary mapping the keyword arguments.
29
30 This specially handles the distinction of `@foo(key = value)` and
31 `@foo(; key = value)` in macros.
32
33 An error is thrown if multiple keyword arguments are passed with the same key.
34
35 If `valid_kwargs` is a `Vector{Symbol}`, an error is thrown if a keyword is not
36 in `valid_kwargs`.
37
38 If `num_positional_args` is not nothing, an error is thrown if the number of
39 positional arguments is not in `num_positional_args`.
40 """
41 function parse_macro_arguments(
42 error_fn::Function,
43 args;
44 valid_kwargs::Union{Nothing,Vector{Symbol}} = nothing,
45 num_positional_args::Union{Nothing,Int,UnitRange{Int}} = nothing,
46 )
47 pos_args, kwargs = Any[], Dict{Symbol,Any}()
48 for arg in _reorder_parameters(args)
49 if Meta.isexpr(arg, :(=), 2)
50 if haskey(kwargs, arg.args[1])
51 error_fn(
52 "the keyword argument `$(arg.args[1])` was given " *
53 "multiple times.",
54 )
55 elseif valid_kwargs !== nothing && !(arg.args[1] in valid_kwargs)
56 error_fn("unsupported keyword argument `$(arg.args[1])`.")
57 elseif !(arg.args[1] isa Symbol)
58 error_fn(
59 "Invalid keyword argument detected. If you are trying to " *
60 "construct an equality constraint, use `==` instead of `=`.",
61 )
62 end
63 kwargs[arg.args[1]] = arg.args[2]
64 else
65 push!(pos_args, arg)
66 end
67 end
68 if num_positional_args isa Int
69 n = length(pos_args)
70 if n != num_positional_args
71 error_fn(
72 "expected $num_positional_args positional arguments, got $n.",
73 )
74 end
75 elseif num_positional_args isa UnitRange{Int}
76 if !(length(pos_args) in num_positional_args)
77 a, b = num_positional_args.start, num_positional_args.stop
78 error_fn(
79 "expected $a to $b positional arguments, got $(length(pos_args)).",
80 )
81 end
82 end
83 return pos_args, kwargs
84 end

158 _parse_ref_sets(expr::Expr)
159
160 Helper function for macros to construct container objects.
161
162 Takes an `Expr` that specifies the container, for example,
163 `:(x[i=1:3,[:red,:blue],k=S; i+k <= 6])`, and returns:
164
165 1. `index_vars`: Names for the index variables, for example, `[:i, gensym(), :k]`.
166 These may also be expressions, like `:((i, j))` from a call like
167 `:(x[(i, j) in S])`.
168 2. `index_sets`: Sets used for indexing, for example, `[1:3, [:red,:blue], S]`
169 3. `condition`: Expr containing any conditional imposed on indexing, or `:()`
170 if none is present
171 """
172 function _parse_ref_sets(error_fn::Function, expr::Expr)
173 c = copy(expr)
174 index_vars, index_sets, condition = Any[], Any[], :()
175 # `:(t[i, j; k])` is a `:ref`, while `:(t[i; j])` is a `:typed_vcat`. In
176 # both cases `:t` is the first argument.
177 if Meta.isexpr(c, :typed_vcat) || Meta.isexpr(c, :ref)
178 name = popfirst!(c.args)
179 if !(name isa Symbol)
180 error_fn(
181 "Unsupported syntax: the expression `$name` cannot be used " *
182 "as a name.",
183 )
184 end
185 end
186 if Meta.isexpr(c, :vcat) || Meta.isexpr(c, :typed_vcat)
187 # An expression like `t[i; k]` or `[i; k]`. The filtering condition is
188 # the second argument.
189 if length(c.args) > 2
190 error_fn(
191 "Unsupported syntax $c: There can be at most one filtering " *
192 "condition, which is separated from the indices by a single " *
193 "`;`.",
194 )
195 elseif length(c.args) == 2
196 condition = pop!(c.args)
197 end
198 # If length(c.args) == 1, expr ends in a trailing `;`, but there is no
199 # condition, so we don't hhave to do anything.
200 elseif Meta.isexpr(c, :ref) || Meta.isexpr(c, :vect)
201 # An expression like `t[i, j; k]` or `[i, j; k]`. The filtering
202 # condition is a `:parameters` expression in the first argument.
203 if Meta.isexpr(c.args[1], :parameters)
204 parameters = popfirst!(c.args)
205 if length(parameters.args) != 1
206 error_fn(
207 "Unsupported syntax $c: There can be at most one " *
208 "filtering condition, which is separated from the " *
209 "indices by a single `;`.",
210 )
211 end
212 condition = parameters.args[1]
213 end
214 end
215 for arg in c.args
216 _parse_index_sets(error_fn, index_vars, index_sets, arg)
217 end
218 return index_vars, index_sets, condition
219 end

411
412 Add the positional arguments `args` to the function call expression `call`,

415 This function is able to incorporate additional positional arguments to `call`s

417 """
418 function add_additional_args(
419 call::Expr,
420 args::Vector,
421 kwargs::Dict{Symbol,Any};
422 kwarg_exclude::Vector{Symbol} = Symbol[],
423 )
424 call_args = call.args
425 if Meta.isexpr(call, :.)
426 # call is broadcasted
427 call_args = call.args[2].args
428 end
429 # Cache all keyword arguments
430 kw_args = filter(Base.Fix2(Meta.isexpr, :kw), call_args)
431 # Remove keyword arguments from the end
432 filter!(!Base.Fix2(Meta.isexpr, :kw), call_args)
433 # Add the new positional arguments
434 append!(call_args, esc.(args))
435 # Re-add the cached keyword arguments back to the end
436 append!(call_args, kw_args)
437 for (key, value) in kwargs
438 if !(key in kwarg_exclude)
439 push!(call_args, esc(Expr(:kw, key, value)))
440 end
441 end
442 return
443 end

455
456 This assumes that the key in `kwargs` used to over-ride the name choice is
457 `:base_name`.
458
459 ## Example
460
461 ```jldoctest
462 julia> Containers.build_name_expr(:x, [:i, :j], Dict{Symbol,Any}())
463 :(string("x", "[", string(\$(Expr(:escape, :i))), ",", string(\$(Expr(:escape, :j))), "]"))
464
465 julia> Containers.build_name_expr(nothing, [:i, :j], Dict{Symbol,Any}())
466 ""
467
468 julia> Containers.build_name_expr(:y, [:i, :j], Dict{Symbol,Any}(:base_name => "y"))
469 :(string("y", "[", string(\$(Expr(:escape, :i))), ",", string(\$(Expr(:escape, :j))), "]"))
470 ```
471 """
472 function build_name_expr(
473 name::Union{Symbol,Nothing},
474 index_vars::Vector,
475 kwargs::Dict{Symbol,Any},
476 )
477 base_name = get(kwargs, :base_name, string(something(name, "")))
478 if !(base_name isa String)
479 base_name = esc(base_name)
480 end
481 if isempty(index_vars) || base_name == ""
482 return base_name
483 end
484 expr = Expr(:call, :string, base_name, "[")
485 for index in index_vars
486 # Converting the arguments to strings before concatenating is faster:
487 # https://github.com/JuliaLang/julia/issues/29550.
488 push!(expr.args, :(string($(esc(index)))))
489 push!(expr.args, ",")
490 end
491 expr.args[end] = "]"
492 return expr
493 end

589 @container([i=..., j=..., ...], expr[, container = :Auto])
590
591 Create a container with indices `i`, `j`, ... and values given by `expr` that
592 may depend on the value of the indices.

602 is transformed to `Base.OneTo(n)` before being given to [`container`](@ref).
603

617

620 Dimension 1, 1:3
621 Dimension 2, 1:3
622 And data, a 3×3 Matrix{Int64}:
623 2 3 4
624 3 4 5
625 4 5 6
626
627 julia> Containers.@container([i = 2:3, j = 1:3], i + j)
628 2-dimensional DenseAxisArray{Int64,2,...} with index sets:
629 Dimension 1, 2:3
630 Dimension 2, Base.OneTo(3)
631 And data, a 2×3 Matrix{Int64}:
632 3 4 5
633 4 5 6
634
635 julia> Containers.@container([i = 1:3, j = 1:3; i <= j], i + j)
636 SparseAxisArray{Int64, 2, Tuple{Int64, Int64}} with 6 entries:
637 [1, 1] = 2
638 [1, 2] = 3
639 [1, 3] = 4
640 [2, 2] = 4
641 [2, 3] = 5
642 [3, 3] = 6
643 ```
644 """
645 macro container(input_args...)

647 error,
648 input_args;

650 valid_kwargs = [:container],
651 )
652 name, index_vars, indices = parse_ref_sets(error, args[1])
653 code = container_code(index_vars, indices, esc(args[2]), kwargs)
654 if name === nothing

656 end
