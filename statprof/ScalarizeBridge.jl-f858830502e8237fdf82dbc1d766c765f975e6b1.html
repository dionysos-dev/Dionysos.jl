7 """
8 ScalarizeBridge{T,F,S}
9
10 `ScalarizeBridge` implements the following reformulations:
11
12 * ``f(x) - a \\in \\mathbb{R}_+`` into ``f_i(x) \\ge a_i`` for all ``i``
13 * ``f(x) - a \\in \\mathbb{R}_-`` into ``f_i(x) \\le a_i`` for all ``i``
14 * ``f(x) - a \\in \\{0\\}`` into ``f_i(x) == a_i`` for all ``i``
15
16 ## Source node
17
18 `ScalarizeBridge` supports:
19
20 * `G` in [`MOI.Nonnegatives{T}`](@ref)
21 * `G` in [`MOI.Nonpositives{T}`](@ref)
22 * `G` in [`MOI.Zeros{T}`](@ref)
23
24 ## Target nodes
25
26 `ScalarizeBridge` creates:
27
28 * `F` in `S`, where `S` is one of [`MOI.GreaterThan{T}`](@ref),
29 [`MOI.LessThan{T}`](@ref), and [`MOI.EqualTo{T}`](@ref), depending on the
30 type of the input set.
31 """

53 function MOI.supports_constraint(
54 ::Type{ScalarizeBridge{T}},
55 F::Type{<:MOI.AbstractVectorFunction},
56 S::Type{<:MOI.Utilities.VectorLinearSet},
57 ) where {T}
58 # If `F` is `MOI.VectorAffineFunction{Complex{Float64}}`, `S` is `MOI.Zeros`
59 # and `T` is `Float64`, it would create a set `MOI.EqualTo{Float64}` which
60 # is incorrect hence we say we only support it if the coefficient type of
61 # `F` is `T`.
62 # If `S` is `MOI.Nonnegatives` or `MOI.Nonpositives` then `T` must be
63 # `Real` as otherwise, `scalar_set_type` will fail in `concrete_bridge_type`.
64 return MOI.Utilities.is_coefficient_type(F, T) &&
65 (S === MOI.Zeros || (T <: Real))
66 end

122 function MOI.get(
123 model::MOI.ModelLike,
124 attr::MOI.ConstraintFunction,
125 bridge::ScalarizeBridge{T},
126 ) where {T}
127 func = MOI.Utilities.vectorize(
128 MOI.get.(model, attr, bridge.scalar_constraints),
129 )
130 if func isa MOI.VectorOfVariables
131 return func
132 end
133 # `func` is in terms of bridged variables here while in `bridge_constraint`
134 # it was in terms of the solver variables so `MOI.constant(set)` might be
135 # different than `bridge.constants[i]`.
136 for i in eachindex(bridge.scalar_constraints)
137 set = MOI.get(model, MOI.ConstraintSet(), bridge.scalar_constraints[i])
138 rhs = MOI.constant(set)
139 if !iszero(rhs)
140 func = MOI.Utilities.operate_output_index!(-, T, i, func, rhs)
141 end
142 end
143 return func
144 end
