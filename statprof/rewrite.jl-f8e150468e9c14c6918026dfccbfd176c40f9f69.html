152 rewrite_generator(expr::Expr, inner::Function)
153
154 Rewrites the generator statements `expr` and returns a properly nested for loop
155 with nested filters as specified.
156
157 # Examples
158
159 ```jldoctest
160 julia> using MutableArithmetics
161
162 julia> MutableArithmetics.rewrite_generator(:(i for i in 1:2 if isodd(i)), i -> :(\$i + 1))
163 :(for \$(Expr(:escape, :(i = 1:2)))
164 if \$(Expr(:escape, :(isodd(i))))
165 i + 1
166 end
167 end)
168 ```

171 # `i + j for i in 1:2 for j in 1:2` is a `flatten` expression

173 return rewrite_generator(ex.args[1], inner)
174 elseif !isexpr(ex, :generator)
175 return inner(ex)

179 if isa(sets, Expr)
180 return sets
181 elseif length(sets) == 1

183 else
184 return Expr(:block, sets...)
185 end
186 end

188 idxvars = []
189 if isexpr(ex.args[2], :filter) # if condition
190 loop = Expr(
191 :for,
192 esc(itrsets(ex.args[2].args[2:end])),
193 Expr(
194 :if,
195 esc(ex.args[2].args[1]),

282 function _is_decomposable_with_factors(ex)
283 # `.+` and `.-` do not support being decomposed if `left_factors` or

291 """
292 rewrite(
293 expr;
294 move_factors_into_sums::Bool = true,
295 return_is_mutable::Bool = false,
296 ) -> Tuple{Symbol,Expr[,Bool]}
297
298 Rewrites the expression `expr` to use mutable arithmetics.
299
300 Returns `(variable, code)` comprised of a `gensym`'d variable equivalent to
301 `expr` and the code necessary to create the variable.
302
303 ## `move_factors_into_sums`
304
305 If `move_factors_into_sums = true`, some terms are rewritten based on the
306 assumption that summations produce a linear function.
307
308 For example, if `move_factors_into_sums = true`, then
309 `y * sum(x[i] for i in 1:2)` is rewritten to:
310 ```julia
311 variable = MA.Zero()
312 for i in 1:2
313 variable = MA.operate!!(MA.add_mul, result, y, x[i])
314 end
315 ```
316
317 If `move_factors_into_sums = false`, it is rewritten to:
318 ```julia
319 term = MA.Zero()
320 for i in 1:2
321 term = MA.operate!!(MA.add_mul, term, x[i])
322 end
323 variable = MA.operate!!(*, y, term)
324 ```
325
326 The latter can produce an additional allocation if there is an efficient
327 fallback for `add_mul` and not for `*(y, term)`.
328
329 ## `return_is_mutable`
330
331 If `return_is_mutable = true`, this function returns three arguments. The third
332 is a `Bool` indicating if the returned expression can be safely mutated without
333 changing the user's original expression.
334
335 `return_is_mutable` cannot be `true` if `move_factors_into_sums = true`.
336 """
337 function rewrite(x; return_is_mutable::Bool = false, kwargs...)
338 variable = gensym()
339 if return_is_mutable
340 code, is_mutable = rewrite_and_return(x; return_is_mutable, kwargs...)
341 return variable, :($variable = $code), is_mutable
342 end
343 code = rewrite_and_return(x; kwargs...)
344 return variable, :($variable = $code)
345 end
