125 function _iterate(map::Map, state = 1)
126 while state ≤ length(map.bridges) && map.bridges[state] === nothing
127 state += 1
128 end
129 if state > length(map.bridges)
130 return _iterate_sv(map)
131 end
132 F, S = map.constraint_types[state]
133 return _index(state, F, S) => map.bridges[state], (1, state + 1)
134 end
