87 """
88 broadcast_mutability(T::Type, ::typeof(op), args::Type...)::MutableTrait
89
90 Return `IsMutable` to indicate an object of type `T` can be modified to be
91 equal to `broadcast(op, args...)`.
92 """
93 function broadcast_mutability(T::Type, op, args::Vararg{Type,N}) where {N}
94 if mutability(T) isa IsMutable && promote_broadcast(op, args...) == T
95 return IsMutable()
96 else
97 return IsNotMutable()
98 end

105 # Some AbstractArray, like JuMP.Containers.SparseAxisArray, do not support
106 # Base.size. In such cases, we default to returning `false`, since we cannot
107 # safely decide whether a broadcast can be stored in `x` unless we know the
108 # sizes of the two entries.
109 function _try_size(x::AbstractArray)

111 return size(x)

113 return missing
114 end

118 _checked_size(x_size::Any, y::AbstractArray) = x_size == _try_size(y)

123 end

127 # This method is a slightly tricky one:
128 #
129 # If the elements in the broadcast are different sized arrays, weird things can
130 # happen during broadcasting since we'll either need to return a different size
131 # to `x`, or multiple copies of an argument will be used for different parts of
132 # `x`. To simplify, let's just return `IsNotMutable` if the sizes are different,
133 # which will be slower but correct.
134 function broadcast_mutability(
135 x::AbstractArray,
136 op,
137 args::Vararg{Any,N},
138 ) where {N}
139 if !_checked_size(_try_size(x), args)::Bool
140 return IsNotMutable()
141 end
142 return broadcast_mutability(typeof(x), op, typeof.(args)...)
143 end

145 broadcast_mutability(::Type) = IsNotMutable()
