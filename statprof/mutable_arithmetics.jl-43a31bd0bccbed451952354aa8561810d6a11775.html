278 # If there are more arguments, we multiply the constants together.
279 @generated function _add_sub_mul_reorder!(
280 op::_MA.AddSubMul,
281 expr::_GenericAffOrQuadExpr,
282 args::Vararg{Any,N},
283 ) where {N}
284 n = length(args)
285 @assert n ≥ 3
286 varidx = findall(t -> (t <: AbstractJuMPScalar), collect(args))
287 allscalar = all(t -> (t <: _Constant), args[setdiff(1:n, varidx)])
288 # We need to get down to only two factors
289 # If there are only constants and one JuMP expressions, then we multiply
290 # the constants together. Otherwise we multiply all factors except the
291 # last one, there may be a better thing to do here.
292 idx = (allscalar && length(varidx) == 1) ? varidx[1] : n
293 coef = Expr(:call, :*, [:(args[$i]) for i in setdiff(1:n, idx)]...)
294 return :(_MA.operate!(op, expr, $coef, args[$idx]))
295 end

308 # `add_mul(expr, args...)` defaults to `muladd(args..., expr)` which gives
309 # `*(args...) + expr`. If `expr isa AbstractJuMPScalar`, this reorders the terms.
310 # The following implementation avoids this issue and is also more efficient.
311 function _MA.add_mul(lhs::AbstractJuMPScalar, x::_Scalar, y::_Scalar)
312 T = _MA.promote_operation(_MA.add_mul, typeof(lhs), typeof(x), typeof(y))
313 expr = _MA.operate(convert, T, lhs)
314 # We can't use `operate!!` here because in the IsNotMutable case (for example,
315 # NonlinearExpr), it will fallback to this method and cause a StackOverflow.
316 if _MA.mutability(T) == _MA.IsNotMutable()
317 return expr + _MA.operate(*, x, y)
318 end
319 return _MA.operate!(_MA.add_mul, expr, x, y)
320 end
