640 # This macro is for expert/internal use only. Prefer the concrete Model type
641 # instantiated below.
642 """
643 macro model(
644 model_name,
645 scalar_sets,
646 typed_scalar_sets,
647 vector_sets,
648 typed_vector_sets,
649 scalar_functions,
650 typed_scalar_functions,
651 vector_functions,
652 typed_vector_functions,
653 is_optimizer = false
654 )
655
656 Creates a type `model_name` implementing the MOI model interface and supporting
657 all combinations of the provided functions and sets.
658
659 Each `typed_` `scalar`/`vector` `sets`/`functions` argument is a tuple of types.
660 A type is "typed" if it has a coefficient `{T}` as the first type parameter.
661
662 ## Tuple syntax
663
664 To give no set/function, write `()`.
665 To give one set or function `X`, write `(X,)`.
666
667 ## `is_optimizer`
668
669 If `is_optimizer = true`, the resulting struct is a of [`GenericOptimizer`](@ref),
670 which is a subtype of [`MOI.AbstractOptimizer`](@ref), otherwise, it is a
671 [`GenericModel`](@ref), which is a subtype of [`MOI.ModelLike`](@ref).
672
673 ## VariableIndex
674
675 * The function [`MOI.VariableIndex`](@ref) must not be given in
676 `scalar_functions`.
677 * The model supports [`MOI.VariableIndex`](@ref)-in-`S` constraints where `S`
678 is [`MOI.EqualTo`](@ref), [`MOI.GreaterThan`](@ref), [`MOI.LessThan`](@ref),
679 [`MOI.Interval`](@ref), [`MOI.Integer`](@ref), [`MOI.ZeroOne`](@ref),
680 [`MOI.Semicontinuous`](@ref) or [`MOI.Semiinteger`](@ref).
681 * The sets supported with [`MOI.VariableIndex`](@ref) cannot be controlled from
682 the macro; use [`UniversalFallback`](@ref) to support more sets.
683
684 ## Example
685
686 The model describing a linear program would be:
687 ```julia
688 @model(
689 LPModel, # model_name
690 (), # untyped scalar sets
691 (MOI.EqualTo, MOI.GreaterThan, MOI.LessThan, MOI.Interval), # typed scalar sets
692 (MOI.Zeros, MOI.Nonnegatives, MOI.Nonpositives), # untyped vector sets
693 (), # typed vector sets
694 (), # untyped scalar functions
695 (MOI.ScalarAffineFunction,), # typed scalar functions
696 (MOI.VectorOfVariables,), # untyped vector functions
697 (MOI.VectorAffineFunction,), # typed vector functions
698 false, # is_optimizer
699 )
700 ```
701 """
702 macro model(
703 model_name,
704 scalar_sets,
705 typed_scalar_sets,
706 vector_sets,
707 typed_vector_sets,
708 scalar_functions,
709 typed_scalar_functions,
710 vector_functions,
711 typed_vector_functions,
712 is_optimizer = false,
713 )
714 scalar_sets = vcat(
715 SymbolSet.(scalar_sets.args, false),
716 SymbolSet.(typed_scalar_sets.args, true),
717 )
718 vector_sets = vcat(
719 SymbolSet.(vector_sets.args, false),
720 SymbolSet.(typed_vector_sets.args, true),
721 )
722 scalar_funs = vcat(
723 SymbolFun.(scalar_functions.args, false),
724 SymbolFun.(typed_scalar_functions.args, true),
725 )
726 vector_funs = vcat(
727 SymbolFun.(vector_functions.args, false),
728 SymbolFun.(typed_vector_functions.args, true),
729 )
730 scname = esc(Symbol("$(model_name)ScalarConstraints"))
731 vcname = esc(Symbol("$(model_name)VectorConstraints"))
732 funs = [scalar_funs; vector_funs]
733 set_struct_types = map(eachindex(funs)) do i
734 cname, sets = if i <= length(scalar_funs)
735 scname, scalar_sets
736 else
737 vcname, vector_sets
738 end
739 voc = map(sets) do set
740 return :(VectorOfConstraints{$(_typed(funs[i])),$(_typed(set))})
741 end
742 return _struct_of_constraints_type(cname, voc, true)
743 end
744 func_name = esc(Symbol("$(model_name)FunctionConstraints"))
745 expr = quote end
746 if length(scalar_sets) >= 2
747 push!(expr.args, struct_of_constraint_code(scname, scalar_sets))
748 end
749 if length(vector_sets) >= 2
750 push!(expr.args, struct_of_constraint_code(vcname, vector_sets))
751 end
752 if length(funs) != 1
753 push!(
754 expr.args,
755 struct_of_constraint_code(func_name, funs, set_struct_types),
756 )
757 end
758 func_typed = _struct_of_constraints_type(func_name, set_struct_types, false)
759 T = esc(:T)
760 generic = is_optimizer ? GenericOptimizer : GenericModel
761 push!(
762 expr.args,
763 quote
764 const $(esc(model_name)){$T} = $generic{
765 $T,
766 ObjectiveContainer{$T},
767 VariablesContainer{$T},
768 $func_typed,
769 }
770 end,
771 )
772 return expr
773 end
